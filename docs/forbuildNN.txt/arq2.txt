def criar_repositorios_ficticios():
    """
    Simula as dimensões reais de dados do PRIDE, CPTAC e TCGA:
    - PRIDE: 120 fosfosítios-alvo monitorados em 6h, 24h, 48h e 72h.
    - CPTAC: 80 características estáticas de expressão proteica de tecido tumoral clínico.
    - TCGA: Características genômicas basais (status de mutação BRAF V600E, níveis de transcritos).

    print("[INFO] Construindo repositórios fictícios isolados para PRIDE, CPTAC e TCGA...")
    
    # 1. PRIDE: Dataset longitudinal de espectrometria de massas de linhagens celulares
    colunas_pride = [f"PRIDE_Amostra_Cel_{r}_t{tp}h" for r in range(10) for tp in [6, 24, 48, 72]]
    df_pride = pd.DataFrame(np.random.uniform(8, 30, size=(120, len(colunas_pride))),
                            index=[f"SitioQuinase_{i}" for i in range(120)], columns=colunas_pride)
    # Injeta padrões típicos de valores ausentes inerentes a fluxos de DIA
    df_pride.iloc[np.random.choice(120, 15, replace=False), np.random.choice(40, 4, replace=False)] = np.nan

    df_cptac.to_csv("dados_origem_cptac_espectrometria.csv")
    
    # 3. TCGA: Estados de mutação genômica e margens globais de expressão de transcritos

# 2. PROCESSAMENTO DE DADOS: LIMPEZA, IMPUTAÇÃO E FORMATAÇÃO DE TENSORES
# ==============================================================================
class DatasetMultiOmicoMelanoma(Dataset):
    def __init__(self, caminho_pride, caminho_cptac, caminho_tcga, caminho_metadata, pontos_temporais=[6, 24, 48, 72]):
        self.pontos_temporais = pontos_temporais
        self.raw_pride = pd.read_csv(caminho_pride, index_col=0)
        self.raw_cptac = pd.read_csv(caminho_cptac, index_col=0)
        self.raw_tcga = pd.read_csv(caminho_tcga, index_col=0)
        self.metadata = pd.read_csv(caminho_metadata)
        
        # Isola e limpa os fluxos de dados de cada repositório para evitar vazamento de informações (data leakage)
        self.processed_pride = self._limpar_matriz_ms(self.raw_pride)
        self.processed_cptac = self._limpar_matriz_ms(self.raw_cptac)
        self.processed_tcga = self._limpar_matriz_genomica(self.raw_tcga)
        
        self.X_pride, self.X_cptac, self.X_tcga, self.y = self._construir_tensores_isolados()

    def _limpar_matriz_ms(self, df):
        # Conversão estrita para Log2 + normalização estrutural pela mediana das colunas
        df_log = np.log2(df.replace(0, np.nan) + 1e-5)
        df_norm = df_log - df_log.median(axis=0)
        # Imputação de valores ausentes por KNN para recuperar linhas espectrais perdidas
        imputer = KNNImputer(n_neighbors=5, weights="distance")
        imputed = imputer.fit_transform(df_norm.T).T
        return pd.DataFrame(imputed, index=df.index, columns=df.columns)

    def _limpar_matriz_genomica(self, df):
        scaler = StandardScaler()
        scaled_features = scaler.fit_transform(df)
        return pd.DataFrame(scaled_features, index=df.index, columns=df.columns)

    def _construir_tensores_isolados(self):
        tensores_pride, tensores_cptac, tensores_tcga, rotulos = [], [], [], []
        agrupados = self.metadata.groupby('id_individuo')
        
        for id_individuo, grupo in agrupados:
            serie_temporal = []
            valido = True
            for tp in self.pontos_temporais:
                correspondencia = grupo[grupo['ponto_temporal'] == tp]['coluna_amostra_pride']
                if not correspondencia.empty and correspondencia.values[0] in self.processed_pride.columns:
                    serie_temporal.append(self.processed_pride[correspondencia.values[0]].values)
                else:
                    valido = False
                    break
            if not valido: continue
                
            col_cptac = grupo.iloc[0]['coluna_amostra_cptac']
            if col_cptac not in self.processed_cptac.columns: continue
            vetor_cptac = self.processed_cptac[col_cptac].values
            
            idx_tcga = grupo.iloc[0]['indice_linha_tcga']
            if idx_tcga not in self.processed_tcga.index: continue
            vetor_tcga = self.processed_tcga.loc[idx_tcga].values
            
            tensores_pride.append(serie_temporal)
            tensores_cptac.append(vetor_cptac)
            tensores_tcga.append(vetor_tcga)
            rotulos.append(grupo[grupo['ponto_temporal'] == 72]['rotulo_resistencia'].values[0])
            
        return (
            torch.tensor(np.array(tensores_pride), dtype=torch.float32),
            torch.tensor(np.array(tensores_cptac), dtype=torch.float32),
            torch.tensor(np.array(tensores_tcga), dtype=torch.float32),
            torch.tensor(np.array(rotulos), dtype=torch.float32).unsqueeze(1)
        )

    def __len__(self): return len(self.y)
    def __getitem__(self, idx): return self.X_pride[idx], self.X_cptac[idx], self.X_tcga[idx], self.y[idx]

# 3. ARQUITETURA CORE: REDE NEURAL DE TRÊS RAMOS COM ATENÇÃO CRUZADA
# ==============================================================================
class RedeMultiRamosMelanoma(nn.Module):
    def __init__(self, dim_pride, dim_cptac, dim_tcga, dim_oculta=32):
        super(RedeMultiRamosMelanoma, self).__init__()
        
        # Ramo 1 (PRIDE): Camada de Cinética Longitudinal Dinâmica (BiLSTM)
        self.pride_lstm = nn.LSTM(input_size=dim_pride, hidden_size=dim_oculta, num_layers=2, batch_first=True, bidirectional=True)
        self.pride_proj = nn.Linear(dim_oculta * 2, dim_oculta)
        self.pride_bn = nn.BatchNorm1d(dim_oculta)
        
        # Ramo 2 (CPTAC): Camada de Abundância Proteômica de Tecido Estático
        self.cptac_fc = nn.Sequential(nn.Linear(dim_cptac, dim_oculta), nn.BatchNorm1d(dim_oculta), nn.ReLU(), nn.Linear(dim_oculta, dim_oculta))
        
        # Ramo 3 (TCGA): Camada de Transcritômica/Mutação Basal do Tumor
        self.tcga_fc = nn.Sequential(nn.Linear(dim_tcga, dim_oculta), nn.BatchNorm1d(dim_oculta), nn.ReLU(), nn.Linear(dim_oculta, dim_oculta))
        
        # Bloco de Atenção Cruzada por Produto Escalar Escalonado de Duas Fontes
        self.camada_query = nn.Linear(dim_oculta, dim_oculta)
        self.key_cptac = nn.Linear(dim_oculta, dim_oculta)
        self.key_tcga = nn.Linear(dim_oculta, dim_oculta)
        self.value_cptac = nn.Linear(dim_oculta, dim_oculta)
        self.value_tcga = nn.Linear(dim_oculta, dim_oculta)
        self.escala = np.sqrt(dim_oculta)
        
        self.classificador = nn.Sequential(
            nn.Linear(dim_oculta * 3, dim_oculta),
            nn.ReLU(),
            nn.Linear(dim_oculta, 1)
        )

    def forward(self, x_pride, x_cptac, x_tcga, retornar_embeddings=False):
        # Extrai características de ramos disjuntos simultaneamente
        lstm_out, _ = self.pride_lstm(x_pride)
        feat_pride = F.relu(self.pride_bn(self.pride_proj(lstm_out[:, -1, :])))
        feat_cptac = F.relu(self.cptac_fc(x_cptac))
        feat_tcga = F.relu(self.tcga_fc(x_tcga))
        
        # Calcula os pesos de atenção cruzada sobre as camadas de background
        Q = self.camada_query(feat_pride).unsqueeze(1)
        K = torch.stack([self.key_cptac(feat_cptac), self.key_tcga(feat_tcga)], dim=1)
        V = torch.stack([self.value_cptac(feat_cptac), self.value_tcga(feat_tcga)], dim=1)
        
        scores = torch.bmm(Q, K.transpose(1, 2)).squeeze(1) / self.escala
        pesos_atencao = F.softmax(scores, dim=-1).unsqueeze(2)
        feat_estatica_atendida = torch.sum(pesos_atencao * V, dim=1)
        
        # Etapa de fusão global tardia de características
        z_fused = torch.cat((feat_pride, feat_cptac, feat_estatica_atendida), dim=1)
        logits = self.classificador(z_fused)
        
        if retornar_embeddings: return logits, z_fused
        return logits

# 5. ORQUESTRADOR DE EXECUÇÃO DO PIPELINE
# ==============================================================================
def executar_pipeline_dissertacao_integrado():
    # 1. Cria os arquivos de cache planos individuais disjuntos
    criar_repositorios_ficticios()
    
    # 2. Pré-processa, limpa e mapeia os repositórios multi-ramos
    dataset = DatasetMultiOmicoMelanoma(
        caminho_pride="dados_origem_pride_dia.csv",
        caminho_cptac="dados_origem_cptac_espectrometria.csv",
        caminho_tcga="dados_origem_tcga_genomica.csv",
        caminho_metadata="mapa_vinculo_clinico_unificado.csv"
    )
    dataloader = DataLoader(dataset, batch_size=2, shuffle=True)
    
    dim_p = dataset.X_pride.shape[2]
    dim_c = dataset.X_cptac.shape[1]
    dim_t = dataset.X_tcga.shape[1]
    
    # 3. Inicializa o Modelo e os Otimizadores
    modelo = RedeMultiRamosMelanoma(dim_pride=dim_p, dim_cptac=dim_c, dim_tcga=dim_t)
    criterio = nn.BCEWithLogitsLoss()
    otimizador = torch.optim.AdamW(modelo.parameters(), lr=0.005, weight_decay=1e-3)
    
    # 4. Loop de Execução da Convergência de Treinamento do Modelo Core
    print("\n[TREINAMENTO] Iniciando sequências de retropropagação (backpropagation) de três ramos...")
    perda_epocas = []
    modelo.train()
    for epoca in range(1, 6):
        perda_total = 0.0
        for x_p, x_c, x_t, y in dataloader:
            otimizador.zero_grad()
            out = modelo(x_p, x_c, x_t)
            perda = criterio(out, y)
            perda.backward()
            otimizador.step()
            perda_total += perda.item() * x_p.size(0)
        perda_media = perda_total / len(dataset)
        perda_epocas.append(perda_media)
        print(f"  Época de Otimização {epoca}/5 | Perda de Entropia Cruzada Cumulativa: {perda_media:.4f}")
        
    # 5. Fase de Avaliação do Modelo e Extração Latente
    modelo.eval()
    probes_todas, alvos_todos, embeddings_todos = [], [], []
    with torch.no_grad():
        for x_p, x_c, x_t, y in dataloader:
            logits, embedding = modelo(x_p, x_c, x_t, retornar_embeddings=True)
            probes_todas.extend(torch.sigmoid(logits).cpu().numpy())
            alvos_todos.extend(y.cpu().numpy())
            embeddings_todos.extend(embedding.cpu().numpy())
            
    y_true = np.array(alvos_todos).squeeze()
    y_prob = np.array(probes_todas).squeeze()
    y_pred = (y_prob >= 0.5).astype(int)
    
    metricas = {
        "Limite da Curva ROC-AUC": roc_auc_score(y_true, y_prob),
        "AUC de Precisão-Revocação": auc(precision_recall_curve(y_true, y_prob)[1], precision_recall_curve(y_true, y_prob)[0]),
        "Pontuação F1 (F1-Score)": f1_score(y_true, y_pred, zero_division=0),
        "Correlação de Matthews": matthews_corrcoef(y_true, y_pred),
        "Coeficiente de Silhueta Latente": silhouette_score(np.array(embeddings_todos), y_true)
    }
    
    # 6. Executa a geração programática e exportação da folha em PDF
    compilar_pdf_vetorial_master(np.array(embeddings_todos), y_true, metricas, perda_epocas)
    
    # Limpa os arquivos temporários do cache para manter a higiene do espaço de trabalho
    for f in ["dados_origem_pride_dia.csv", "dados_origem_cptac_espectrometria.csv", "dados_origem_tcga_genomica.csv", "mapa_vinculo_clinico_unificado.csv"]:
        if os.path.exists(f): os.remove(f)
    print("[INFO] Arquivos temporários removidos com segurança. Execução finalizada.\n")

if __name__ == "__main__":
    executar_pipeline_dissertacao_integrado()
